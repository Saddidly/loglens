cmake_minimum_required(VERSION 3.20)
project(loglens VERSION 1.0.0 LANGUAGES C)
add_library(loglens_core src/analysis.c src/report.c)
target_include_directories(loglens_core PUBLIC include)
set_target_properties(loglens_core PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
  target_compile_definitions(loglens_core PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(loglens_core PRIVATE /W4 /WX)
else()
  target_compile_options(loglens_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(loglens src/main.c)
target_link_libraries(loglens PRIVATE loglens_core)
if(MINGW)
  target_link_options(loglens PRIVATE -municode)
endif()
if(MSVC)
  target_compile_definitions(loglens PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME behavioral COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:loglens>)
endif()
install(TARGETS loglens RUNTIME DESTINATION bin)
